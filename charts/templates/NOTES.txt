==============================================================================
 MailBaby Email Service is successfully deployed on Kubernetes!
==============================================================================

Release Name: {{ .Release.Name }}
Namespace:    {{ .Release.Namespace }}

1. Get the application status:
  kubectl get pods -n {{ .Release.Namespace }} -l "{{ include "mailbaby.selectorLabels" . | replace ": " "=" }}"

2. Access the HTTP REST API & Health Probes:
{{- if .Values.ingress.enabled }}
  {{- range $host := .Values.ingress.hosts }}
  http{{ if $.Values.ingress.tls }}s{{ end }}://{{ $host.host }}
  {{- end }}
{{- else if contains "NodePort" .Values.service.type }}
  export NODE_PORT=$(kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[?(@.name=='http')].nodePort}" services {{ include "mailbaby.fullname" . }})
  export NODE_IP=$(kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
  echo http://$NODE_IP:$NODE_PORT
{{- else if contains "LoadBalancer" .Values.service.type }}
  NOTE: It may take a few minutes for the LoadBalancer IP to be available.
  export SERVICE_IP=$(kubectl get svc --namespace {{ .Release.Namespace }} {{ include "mailbaby.fullname" . }} --template "{{"{{ range (index .status.loadBalancer.ingress 0) }}{{.}}{{ end }}"}}")
  echo http://$SERVICE_IP:{{ .Values.service.httpPort }}
{{- else }}
  echo "Forwarding HTTP port 8080 locally..."
  kubectl port-forward --namespace {{ .Release.Namespace }} svc/{{ include "mailbaby.fullname" . }} 8080:{{ .Values.service.httpPort }}

  # Test health endpoints:
  curl http://127.0.0.1:8080/livez
  curl http://127.0.0.1:8080/readyz
  curl http://127.0.0.1:8080/metrics
{{- end }}

3. Send a test email via HTTP REST API:
  curl -X POST http://127.0.0.1:8080/v1/email/send \
    -H "Content-Type: application/json" \
    {{- if .Values.config.auth.enabled }}
    -H "{{ .Values.config.auth.header_name | default "X-API-Key" }}: <YOUR_SECRET_KEY>" \
    {{- end }}
    -d '{
      "to": ["recipient@example.com"],
      "subject": "Hello from MailBaby Kubernetes",
      "text_body": "This is a test email sent via MailBaby HTTP REST API."
    }'

4. Access the gRPC Email Service:
  echo "Forwarding gRPC port 8081 locally..."
  kubectl port-forward --namespace {{ .Release.Namespace }} svc/{{ include "mailbaby.fullname" . }} 8081:{{ .Values.service.grpcPort }}

  # Ping the gRPC service (using grpcurl):
  grpcurl -plaintext 127.0.0.1:8081 mailbaby.v1.MailService/Ping

  # Send an email via gRPC:
  grpcurl -plaintext \
    {{- if .Values.config.auth.enabled }}
    -H "{{ .Values.config.auth.header_name | default "X-API-Key" }}: <YOUR_SECRET_KEY>" \
    {{- end }}
    -d '{
      "to": ["recipient@example.com"],
      "subject": "Hello from MailBaby gRPC",
      "text_body": "This is a test email sent via MailBaby gRPC."
    }' \
    127.0.0.1:8081 mailbaby.v1.MailService/Send
